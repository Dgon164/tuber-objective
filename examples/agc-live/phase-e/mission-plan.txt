# phase asks for all three turn rates at zero (f3, f4, f5), not the pitch rate alone. Phase D's loop only ever
# pressed +pitch and -pitch, so one rate was enough; this loop presses roll and yaw too, and met the pitch
# target while still turning on the other axes.

staged:     f17 == 1 ; f16 == 1
orbit:      f20 > 32000 ; f19 < 3000 ; f16 == 0
